; LPC2148 - LAB QUESTION 12: Switch_P0_14_to_LED_P0_16
; Vector table is at flash address 0, with LPC2148 checksum at 0x14.
; No vendor startup.s should be included in the same Keil target.
        AREA    RESET, CODE, READONLY
        ARM
        ENTRY
Vectors
        B       Reset_Entry
        B       Default_Hang
        B       Default_Hang
        B       Default_Hang
        B       Default_Hang
        DCD     0x99FFFFE2
        B       IRQ_Entry
        B       Default_Hang
Reset_Entry
        B       Main
IRQ_Entry
        B       Default_Hang
Default_Hang
        B       Default_Hang
Main
        ; Switch P0.14 to GND with external pull-up to 3.3 V.
        ; LED P0.16 active LOW; pressed=ON.
        LDR     R0, =0xE002C000    ; PINSEL0
        LDR     R1, [R0]
        LDR     R2, =0x30000000
        BIC     R1, R1, R2        ; GPIO P0.14
        STR     R1, [R0]
        LDR     R0, =0xE002C004    ; PINSEL1
        LDR     R1, [R0]
        BIC     R1, R1, #3       ; GPIO P0.16
        STR     R1, [R0]
        LDR     R4, =0x00010000
        LDR     R0, =0xE0028004
        STR     R4, [R0]        ; LED initially off
        LDR     R0, =0xE0028008
        LDR     R1, [R0]
        BIC     R1, R1, #0x4000 ; P0.14 input
        ORR     R1, R1, R4      ; P0.16 output
        STR     R1, [R0]
Switch_Loop
        LDR     R0, =0xE0028000
        LDR     R1, [R0]
        TST     R1, #0x4000
        BNE     Button_Released
        LDR     R0, =0xE002800C ; LED on
        STR     R4, [R0]
        B       Switch_Loop
Button_Released
        LDR     R0, =0xE0028004 ; LED off
        STR     R4, [R0]
        B       Switch_Loop

        LTORG
        END
