; LPC2148 - LAB QUESTION 18: Internal_DAC_10bit_ramp_P0_25
; Vector table is at flash address 0, with LPC2148 checksum at 0x14.
; No vendor startup.s should be included in the same Keil target.
        AREA    RESET, CODE, READONLY
        ARM
        ENTRY
Vectors
        B       Reset_Entry
        B       Default_Hang
        B       Default_Hang
        B       Default_Hang
        B       Default_Hang
        DCD     0x99FFFFE2
        B       IRQ_Entry
        B       Default_Hang
Reset_Entry
        B       Main
IRQ_Entry
        B       Default_Hang
Default_Hang
        B       Default_Hang
Main
        ; DAC analog output on P0.25 / AOUT.
        ; Assumes VREF is connected to 3.3 V on LPC2148 board.
        LDR     R0, =0xE002C004 ; PINSEL1
        LDR     R1, [R0]
        LDR     R2, =0x000C0000 ; P0.25 mux bits [19:18]
        BIC     R1, R1, R2
        ORR     R1, R1, #0x80000 ; select AOUT = 10b
        STR     R1, [R0]
        LDR     R0, =0xE006C000 ; DACR
        MOV     R4, #0
DAC_Ramp_Loop
        MOV     R5, R4, LSL #6  ; DACR VALUE field [15:6]
        STR     R5, [R0]
        BL      Short_Delay
        ADD     R4, R4, #1
        CMP     R4, #1024
        MOVEQ   R4, #0
        B       DAC_Ramp_Loop
Short_Delay
        MOV     R8, #60         ; adjust to change ramp rate
Ramp_Delay_Loop
        SUBS    R8, R8, #1
        BNE     Ramp_Delay_Loop
        BX      LR

        LTORG
        END
