; LPC2148 - LAB QUESTION 02: Ring_counter_8_LEDs_active_low
; Vector table is at flash address 0, with LPC2148 checksum at 0x14.
; No vendor startup.s should be included in the same Keil target.
        AREA    RESET, CODE, READONLY
        ARM
        ENTRY
Vectors
        B       Reset_Entry
        B       Default_Hang
        B       Default_Hang
        B       Default_Hang
        B       Default_Hang
        DCD     0x99FFFFE2
        B       IRQ_Entry
        B       Default_Hang
Reset_Entry
        B       Main
IRQ_Entry
        B       Default_Hang
Default_Hang
        B       Default_Hang
Main
        ; Select GPIO for P0.16..P0.23 while preserving upper PINSEL1.
        LDR     R0, =0xE002C004    ; PINSEL1
        LDR     R1, [R0]
        LDR     R2, =0xFFFF0000
        AND     R1, R1, R2
        STR     R1, [R0]
        LDR     R0, =0xE0028004    ; IO0SET
        LDR     R1, =0x00FF0000
        STR     R1, [R0]           ; LEDs OFF before output enable
        LDR     R0, =0xE0028008    ; IO0DIR
        LDR     R2, [R0]
        ORR     R2, R2, R1
        STR     R2, [R0]
        ; Exactly one of the eight active-low LEDs is ON.
        MOV     R4, #1
Ring_Loop
        LDR     R0, =0xE0028004    ; IO0SET: all off
        LDR     R1, =0x00FF0000
        STR     R1, [R0]
        LDR     R0, =0xE002800C    ; IO0CLR: one LED on
        MOV     R2, R4, LSL #16
        STR     R2, [R0]
        BL      Delay
        MOV     R4, R4, LSL #1
        CMP     R4, #256
        MOVEQ   R4, #1
        B       Ring_Loop
Delay
        LDR     R9, =300000      ; software delay, speed depends on clock
Delay_Loop
        SUBS    R9, R9, #1
        BNE     Delay_Loop
        BX      LR

        LTORG
        END
