; LPC2148 - LAB QUESTION 10: Seven_segment_P0_16_P0_22
; Vector table is at flash address 0, with LPC2148 checksum at 0x14.
; No vendor startup.s should be included in the same Keil target.
        AREA    RESET, CODE, READONLY
        ARM
        ENTRY
Vectors
        B       Reset_Entry
        B       Default_Hang
        B       Default_Hang
        B       Default_Hang
        B       Default_Hang
        DCD     0x99FFFFE2
        B       IRQ_Entry
        B       Default_Hang
Reset_Entry
        B       Main
IRQ_Entry
        B       Default_Hang
Default_Hang
        B       Default_Hang
Main
        ; Select GPIO for P0.16..P0.23 while preserving upper PINSEL1.
        LDR     R0, =0xE002C004    ; PINSEL1
        LDR     R1, [R0]
        LDR     R2, =0xFFFF0000
        AND     R1, R1, R2
        STR     R1, [R0]
        LDR     R0, =0xE0028004    ; IO0SET
        LDR     R1, =0x00FF0000
        STR     R1, [R0]           ; LEDs OFF before output enable
        LDR     R0, =0xE0028008    ; IO0DIR
        LDR     R2, [R0]
        ORR     R2, R2, R1
        STR     R2, [R0]
        ; Common-cathode 7-seg via drivers/resistors.
        ; P0.16=a, 17=b, ... P0.22=g; P0.23=DP(always off).
        ; This experiment uses active-HIGH segment signals,
        ; unlike question 2 LEDs (which are active-LOW).
        MOV     R4, #0
Display_Loop
        LDR     R5, =Seg_Table
        LDRB    R6, [R5, R4]
        MOV     R6, R6, LSL #16
        LDR     R0, =0xE002800C
        LDR     R1, =0x00FF0000
        STR     R1, [R0]          ; segments OFF
        LDR     R0, =0xE0028004
        STR     R6, [R0]          ; enable required segments
        BL      Delay
        ADD     R4, R4, #1
        CMP     R4, #10
        MOVEQ   R4, #0
        B       Display_Loop
Delay
        LDR     R9, =300000      ; software delay, speed depends on clock
Delay_Loop
        SUBS    R9, R9, #1
        BNE     Delay_Loop
        BX      LR
        ALIGN
Seg_Table
        DCB     0x3F, 0x06, 0x5B, 0x4F, 0x66
        DCB     0x6D, 0x7D, 0x07, 0x7F, 0x6F

        LTORG
        END
