; LPC2148 - LAB QUESTION 11: Logical_operations_memory
; Vector table is at flash address 0, with LPC2148 checksum at 0x14.
; No vendor startup.s should be included in the same Keil target.
        AREA    RESET, CODE, READONLY
        ARM
        ENTRY
Vectors
        B       Reset_Entry
        B       Default_Hang
        B       Default_Hang
        B       Default_Hang
        B       Default_Hang
        DCD     0x99FFFFE2
        B       IRQ_Entry
        B       Default_Hang
Reset_Entry
        B       Main
IRQ_Entry
        B       Default_Hang
Default_Hang
        B       Default_Hang
Main
        ; A=0xAA, B=0x0F.
        ; RAM[8] AND, [12] OR, [16] XOR, [20] A AND NOT B,
        ; RAM[24] 32-bit NOT A.
        LDR     R0, =0x40000000
        MOV     R1, #0xAA
        MOV     R2, #0x0F
        STR     R1, [R0]
        STR     R2, [R0, #4]
        AND     R3, R1, R2
        STR     R3, [R0, #8]
        ORR     R3, R1, R2
        STR     R3, [R0, #12]
        EOR     R3, R1, R2
        STR     R3, [R0, #16]
        BIC     R3, R1, R2
        STR     R3, [R0, #20]
        MVN     R3, R1
        STR     R3, [R0, #24]
Stop
        B       Stop

        LTORG
        END
